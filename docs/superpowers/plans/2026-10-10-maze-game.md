**Goal:** 在 FlowKit 中新增「迷境」模块：摇杆控制小球走迷宫（4 档难度、确定性种子、镜头系统、彩虹轨迹、历史记录），技术栈为 react-native-svg 静态层 + 原生 View 动态层的混合架构。

**Tech Stack:** React Native 0.86（新架构）、TypeScript strict、react-native-svg（新增依赖）、Zustand、AsyncStorage、Jest + react-test-renderer。

```ts
/** mulberry32：确定性伪随机数生成器（全部随机性的唯一来源） */
export function mulberry32(a: number): () => number {
  return function () {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
```

```ts
import type { ShapeKey } from '../types';

/** 形状多边形：坐标 (u, v) ∈ [-1,1]，u 向右、v 向上 */
export type Polygon = [number, number][];

function rad(d: number): number {
  return (d * Math.PI) / 180;
}

export function circlePolygon(): Polygon {
  const pts: Polygon = [];
  for (let i = 0; i < 120; i++) {
    const th = (i / 120) * Math.PI * 2;
    pts.push([0.92 * Math.cos(th), 0.92 * Math.sin(th)]);
  }
  return pts;
}

export function ellipsePolygon(): Polygon {
  const pts: Polygon = [];
  for (let i = 0; i < 120; i++) {
    const th = (i / 120) * Math.PI * 2;
    pts.push([0.95 * Math.cos(th), 0.72 * Math.sin(th)]);
  }
  return pts;
}

export function capsulePolygon(): Polygon {
  const pts: Polygon = [];
  for (let i = 0; i <= 40; i++) {
    const t = -Math.PI / 2 + (Math.PI * i) / 40;
    pts.push([0.62 + 0.5 * Math.cos(t), 0.5 * Math.sin(t)]);
  }
  for (let i = 0; i <= 40; i++) {
    const t = Math.PI / 2 + (Math.PI * i) / 40;
    pts.push([-0.62 + 0.5 * Math.cos(t), 0.5 * Math.sin(t)]);
  }
  return pts;
}

export function hexagonPolygon(): Polygon {
  const pts: Polygon = [];
  for (let i = 0; i < 6; i++) {
    const th = rad(-90 + 60 * i + 30);
    pts.push([0.9 * Math.cos(th), 0.9 * Math.sin(th)]);
  }
  return pts;
}

export function heartPolygon(): Polygon {
  const pts: Polygon = [];
  for (let i = 0; i < 240; i++) {
    const t = (i / 240) * Math.PI * 2;
    const x = 16 * Math.pow(Math.sin(t), 3);
    const y = 13 * Math.cos(t) - 5 * Math.cos(2 * t) - 2 * Math.cos(3 * t) - Math.cos(4 * t);
    pts.push([x / 16, (y + 17) / 14.6 - 1]);
  }
  return pts;
}

export function starPolygon(): Polygon {
  const pts: Polygon = [];
  for (let i = 0; i < 10; i++) {
    const r = i % 2 === 0 ? 0.95 : 0.42;
    const th = -Math.PI / 2 + (i * Math.PI) / 5;
    pts.push([r * Math.cos(th), r * Math.sin(th)]);
  }
  return pts;
}

export function flowerPolygon(): Polygon {
  const pts: Polygon = [];
  for (let i = 0; i < 240; i++) {
    const th = (i / 240) * Math.PI * 2;
    const r = 0.62 + 0.33 * Math.cos(5 * th);
    pts.push([r * Math.cos(th), r * Math.sin(th)]);
  }
  return pts;
}

export function crescentPolygon(): Polygon {
  const pts: Polygon = [];
  for (let i = 0; i <= 60; i++) {
    const th = 57.5 + ((302.5 - 57.5) * i) / 60;
    pts.push([0.92 * Math.cos(rad(th)), 0.92 * Math.sin(rad(th))]);
  }
  for (let i = 0; i <= 40; i++) {
    const b = 284 - ((284 - 76) * i) / 40;
    pts.push([0.3 + 0.8 * Math.cos(rad(b)), 0.8 * Math.sin(rad(b))]);
  }
  return pts;
}

export function organicPolygon(rng: () => number): Polygon {
  const a1 = rng() * Math.PI * 2;
  const a2 = rng() * Math.PI * 2;
  const a3 = rng() * Math.PI * 2;
  const pts: Polygon = [];
  for (let i = 0; i < 240; i++) {
    const th = (i / 240) * Math.PI * 2;

/** 射线法点入判定 */
export function pointInPolygon(u: number, v: number, poly: Polygon): boolean {
  let inside = false;
  for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
    const xi = poly[i][0];
    const yi = poly[i][1];
    const xj = poly[j][0];
    const yj = poly[j][1];
    if ((yi > v) !== (yj > v) && u < ((xj - xi) * (v - yi)) / (yj - yi) + xi) {
      inside = !inside;
    }
  }
  return inside;
}
```

```ts
import { DIFFS } from '../constants';
import type { CellPoint, MazeData, MazeDifficulty, ShapeKey } from '../types';
import { mulberry32 } from './rng';
import { makeShape, pointInPolygon, type Polygon } from './shapes';

/** 形状网格：仅奇数坐标（格）做点入判定 */
export function buildShapeGrid(poly: Polygon, N: number): boolean[][] {
  const G = 2 * N + 1;
  const inS: boolean[][] = [];
  for (let r = 0; r < G; r++) {
    const row: boolean[] = [];
    for (let c = 0; c < G; c++) {
      if (r % 2 === 1 && c % 2 === 1) {
        const u = (2 * c) / (G - 1) - 1;
        const v = 1 - (2 * r) / (G - 1);
        row.push(pointInPolygon(u, v, poly));
      } else {
        row.push(false);
      }
    }
    inS.push(row);
  }
  return inS;
}

/** 死路格计数（开放通道数 = 1，通道打开以两格之间的墙单元打通为准） */
export function countDeadEnds(cells: number[][], inS: boolean[][], G: number): number {
  const N = (G - 1) / 2;
  const dirs = [
    [1, 0],
    [-1, 0],
    [0, 1],
    [0, -1],
  ];
  let count = 0;
  for (let i = 0; i < N; i++) {
    for (let j = 0; j < N; j++) {
      if (!inS[2 * j + 1][2 * i + 1] || cells[2 * j + 1][2 * i + 1] !== 0) continue;
      let open = 0;
      for (const [di, dj] of dirs) {
        const ni = i + di;
        const nj = j + dj;
        if (ni < 0 || nj < 0 || ni >= N || nj >= N) continue;
        if (cells[2 * j + 1 + dj][2 * i + 1 + di] === 0) open++;
      }
      if (open === 1) count++;
    }
  }
  return count;
}

/** braid：按比例随机打通死路格的"对面在形状内"的墙，形成环路 */
export function braidMaze(
  cells: number[][],
  inS: boolean[][],
  G: number,
  rng: () => number,
  ratio: number,
): void {
  const N = (G - 1) / 2;
  const dirs = [
    [1, 0],
    [-1, 0],
    [0, 1],
    [0, -1],
  ];
  const deadEnds: CellPoint[] = [];
  for (let i = 0; i < N; i++) {
    for (let j = 0; j < N; j++) {
      if (!inS[2 * j + 1][2 * i + 1] || cells[2 * j + 1][2 * i + 1] !== 0) continue;
      let open = 0;
      for (const [di, dj] of dirs) {
        const ni = i + di;
        const nj = j + dj;
        if (ni < 0 || nj < 0 || ni >= N || nj >= N) continue;
        if (cells[2 * j + 1 + dj][2 * i + 1 + di] === 0) open++;
      }
      if (open === 1) deadEnds.push([i, j]);
    }
  }
  for (let i = deadEnds.length - 1; i > 0; i--) {
    const k = Math.floor(rng() * (i + 1));
    const tmp = deadEnds[i];
    deadEnds[i] = deadEnds[k];
    deadEnds[k] = tmp;
  }
  const count = Math.floor(deadEnds.length * ratio);
  for (let n = 0; n < count; n++) {
    const [i, j] = deadEnds[n];
    const cands: CellPoint[] = []; // 候选墙单元网格坐标 [wc, wr]
    for (const [di, dj] of dirs) {
      const ni = i + di;
      const nj = j + dj;
      if (ni < 0 || nj < 0 || ni >= N || nj >= N) continue;
      if (!inS[2 * nj + 1][2 * ni + 1]) continue; // 对面格须在形状内（原型行为）
      const wr = 2 * j + 1 + dj;
      const wc = 2 * i + 1 + di;
      if (wr > 0 && wr < G - 1 && wc > 0 && wc < G - 1 && cells[wr][wc] === 1) {
        cands.push([wc, wr]);
      }
    }
    if (cands.length) {
      const [wc, wr] = cands[Math.floor(rng() * cands.length)];
      cells[wr][wc] = 0;
    }
  }
}

/** 格坐标 BFS 最远点（只走 cells=0 的格与已打开通道） */
export function bfsFarthest(cells: number[][], N: number, from: CellPoint): CellPoint {
  const keyOf = (i: number, j: number) => i + ',' + j;
  const dist: Record<string, number> = { [keyOf(from[0], from[1])]: 0 };
  const q: CellPoint[] = [from];
  let far = from;
  let maxd = 0;
  const dirs = [
    [1, 0],
    [-1, 0],
    [0, 1],
    [0, -1],
  ];
  while (q.length) {
    const cur = q.shift()!;
    const d = dist[keyOf(cur[0], cur[1])];
    if (d > maxd) {
      maxd = d;
      far = cur;
    }
    for (const [di, dj] of dirs) {
      const ni = cur[0] + di;
      const nj = cur[1] + dj;
      if (ni < 0 || nj < 0 || ni >= N || nj >= N) continue;
      if (cells[2 * nj + 1][2 * ni + 1] !== 0) continue;
      if (cells[2 * cur[1] + 1 + dj][2 * cur[0] + 1 + di] !== 0) continue;
      const k = keyOf(ni, nj);
      if (k in dist) continue;
      dist[k] = d + 1;
      q.push([ni, nj]);
    }
  }
  return far;
}

/** 完整生成：种子 + 难度 → 确定性迷宫（随机消费顺序固定，不可调换） */
export function generateMaze(seed: number, difficulty: MazeDifficulty): MazeData {
  const cfg = DIFFS[difficulty];
  const N = cfg.N;
  const G = 2 * N + 1;
  const rng = mulberry32(seed);
  const shapeKey: ShapeKey = cfg.pool[Math.floor(rng() * cfg.pool.length)];
  const poly = makeShape(shapeKey, rng);
  const inS = buildShapeGrid(poly, N);

  const cells: number[][] = [];
  for (let r = 0; r < G; r++) cells.push(new Array<number>(G).fill(1));

  // 起点：距形状中心最近的形状内格
  let startCell: CellPoint = [0, 0];
  let bestD = Infinity;
  for (let r = 1; r < G; r += 2) {
    for (let c = 1; c < G; c += 2) {
      if (!inS[r][c]) continue;
      const u = (2 * c) / (G - 1) - 1;
      const v = 1 - (2 * r) / (G - 1);
      const d = u * u + v * v;
      if (d < bestD) {
        bestD = d;
        startCell = [(c - 1) / 2, (r - 1) / 2];
      }
    }
  }

  // DFS 迭代回溯：仅在形状内格之间打通
  const sc = startCell;
  const stack: CellPoint[] = [sc];
  cells[2 * sc[1] + 1][2 * sc[0] + 1] = 0;
  const gridDirs = [
    [0, 2],
    [0, -2],
    [2, 0],
    [-2, 0],
  ];
  while (stack.length) {
    const cur = stack[stack.length - 1];
    const gr = 2 * cur[1] + 1;
    const gc = 2 * cur[0] + 1;
    const opts: CellPoint[] = [];
    for (const [dr, dc] of gridDirs) {
      const nr = gr + dr;
      const nc = gc + dc;
      if (nr > 0 && nr < G - 1 && nc > 0 && nc < G - 1 && cells[nr][nc] === 1 && inS[nr][nc]) {
        opts.push([(nc - 1) / 2, (nr - 1) / 2]);
      }
    }
    if (!opts.length) {
      stack.pop();
      continue;
    }
    const pick = opts[Math.floor(rng() * opts.length)];
    cells[cur[1] + pick[1] + 1][cur[0] + pick[0] + 1] = 0;
    cells[2 * pick[1] + 1][2 * pick[0] + 1] = 0;
    stack.push(pick);
  }

  if (cfg.braid > 0) braidMaze(cells, inS, G, rng, cfg.braid);

  const a = bfsFarthest(cells, N, sc);
  const b = bfsFarthest(cells, N, a);
  return { seed, difficulty, N, G, shapeKey, cells, inS, start: a, end: b };
}
```

```ts
import { BALL_R } from '../constants';
import type { BallPos, MazeData } from '../types';

/** 矩形是否与任何墙单元相交（世界单位） */
export function collidesRect(m: MazeData, x0: number, y0: number, x1: number, y1: number): boolean {
  const c0 = Math.floor(x0);
  const c1 = Math.floor(x1);
  const r0 = Math.floor(y0);
  const r1 = Math.floor(y1);
  for (let r = r0; r <= r1; r++) {
    for (let c = c0; c <= c1; c++) {
      if (c < 0 || r < 0 || c >= m.G || r >= m.G) return true;
      if (m.cells[r][c] === 1) return true;
    }
  }
  return false;
}

/** 单轴移动：畅通直行，否则二分逼近钳制到接触位置（撞墙滑行） */
function moveAxis(m: MazeData, ball: BallPos, axis: 'x' | 'y', amount: number): void {
  if (!amount) return;
  const test = (d: number): boolean => {
    const nx = axis === 'x' ? ball.x + d : ball.x;
    const ny = axis === 'y' ? ball.y + d : ball.y;
    return collidesRect(m, nx - BALL_R, ny - BALL_R, nx + BALL_R, ny + BALL_R);
  };
  if (!test(amount)) {
    if (axis === 'x') ball.x += amount;
    else ball.y += amount;
    return;
  }
  let lo = 0;
  let hi = amount;
  for (let k = 0; k < 8; k++) {
    const mid = (lo + hi) / 2;
    if (test(mid)) hi = mid;
    else lo = mid;
  }
  if (axis === 'x') ball.x += lo;
  else ball.y += lo;
}

/** 按位移移动小球（轴分离 + 滑行），原地修改 ball */
export function moveBall(m: MazeData, ball: BallPos, dx: number, dy: number): void {
  moveAxis(m, ball, 'x', dx);
  moveAxis(m, ball, 'y', dy);
}
```

```ts
import type { CellPoint } from '../types';

const B64_CHARS = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/';

function bytesToBase64(bytes: number[]): string {
  let out = '';
  for (let i = 0; i < bytes.length; i += 3) {
    const b0 = bytes[i];
    const b1 = i + 1 < bytes.length ? bytes[i + 1] : 0;
    const b2 = i + 2 < bytes.length ? bytes[i + 2] : 0;
    out += B64_CHARS[b0 >> 2];
    out += B64_CHARS[((b0 & 3) << 4) | (b1 >> 4)];
    out += i + 1 < bytes.length ? B64_CHARS[((b1 & 15) << 2) | (b2 >> 6)] : '=';
    out += i + 2 < bytes.length ? B64_CHARS[b2 & 63] : '=';
  }
  return out;
}

function base64ToBytes(s: string): number[] {
  const bytes: number[] = [];
  const clean = s.replace(/=+$/, '');
  for (let i = 0; i < clean.length; i += 4) {
    const c0 = B64_CHARS.indexOf(clean[i]);
    const c1 = i + 1 < clean.length ? B64_CHARS.indexOf(clean[i + 1]) : 0;
    const c2 = i + 2 < clean.length ? B64_CHARS.indexOf(clean[i + 2]) : 0;
    const c3 = i + 3 < clean.length ? B64_CHARS.indexOf(clean[i + 3]) : 0;
    bytes.push((c0 << 2) | (c1 >> 4));
    if (i + 2 < clean.length) bytes.push(((c1 & 15) << 4) | (c2 >> 2));
    if (i + 3 < clean.length) bytes.push(((c2 & 3) << 6) | c3);
  }
  return bytes;
}

/**
 * 编码轨迹：16bit 格数 + 8bit 起始 i + 8bit 起始 j + 每步 2bit 方向
 * 方向：0 上(j-1) / 1 右(i+1) / 2 下(j+1) / 3 左(i-1)
 * 一步约 0.25 字节 → 千步约 0.3KB
 */
export function encodeTrail(cells: CellPoint[]): string {
  if (!cells.length) return '';
  const bits: number[] = [];
  const pushN = (val: number, n: number) => {
    for (let b = n - 1; b >= 0; b--) bits.push((val >> b) & 1);
  };
  pushN(cells.length, 16);
  pushN(cells[0][0], 8);
  pushN(cells[0][1], 8);
  for (let n = 1; n < cells.length; n++) {
    const di = cells[n][0] - cells[n - 1][0];
    const dj = cells[n][1] - cells[n - 1][1];
    const dir = dj === 1 ? 2 : dj === -1 ? 0 : di === 1 ? 1 : 3;
    pushN(dir, 2);
  }
  const bytes: number[] = [];
  for (let i = 0; i < bits.length; i += 8) {
    let byte = 0;
    for (let b = 0; b < 8; b++) byte = (byte << 1) | (bits[i + b] || 0);
    bytes.push(byte);
  }
  return bytesToBase64(bytes);
}

export function decodeTrail(s: string): CellPoint[] {
  if (!s) return [];
  const bytes = base64ToBytes(s);
  const bits: number[] = [];
  for (const v of bytes) {
    for (let b = 7; b >= 0; b--) bits.push((v >> b) & 1);
  }
  let pos = 0;
  const readN = (n: number): number => {
    let val = 0;
    for (let b = 0; b < n; b++) {
      val = (val << 1) | (bits[pos] || 0);
      pos++;
    }
    return val;
  };
  const count = readN(16);
  // 自洽校验：合法编码的位长恰为 32 + 2*(count-1)，字节数须恰好容纳且填充位 < 8；
  // 不满足即为损坏输入，安全返回空数组
  const needBits = 32 + 2 * (count - 1);
  if (count < 1 || needBits > bytes.length * 8 || needBits + 8 <= bytes.length * 8) return [];
  const list: CellPoint[] = [[readN(8), readN(8)]];
  for (let n = 1; n < count; n++) {
    const d = readN(2);
    const prev = list[list.length - 1];
    list.push([prev[0] + (d === 1 ? 1 : d === 3 ? -1 : 0), prev[1] + (d === 2 ? 1 : d === 0 ? -1 : 0)]);
  }
  return list;
}
```

```ts
import AsyncStorage from '@react-native-async-storage/async-storage';
import { HISTORY_LIMIT, STORAGE_BEST, STORAGE_HISTORY } from '../constants';
import type { BestRecord, HistoryRecord, MazeDifficulty } from '../types';

export type BestMap = Partial<Record<MazeDifficulty, BestRecord>>;

export async function loadBest(): Promise<BestMap> {
  try {
    const raw = await AsyncStorage.getItem(STORAGE_BEST);
    if (!raw) return {};
    const parsed: unknown = JSON.parse(raw);
    if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) return {};
    return parsed as BestMap;
  } catch {
    return {};
  }
}

/** 仅当更快时覆盖；返回是否刷新了记录 */
export async function upsertBest(difficulty: MazeDifficulty, rec: BestRecord): Promise<boolean> {
  const best = await loadBest();
  const prev = best[difficulty];
  if (prev && prev.timeMs <= rec.timeMs) return false;
  best[difficulty] = rec;
  await AsyncStorage.setItem(STORAGE_BEST, JSON.stringify(best));
  return true;
}

function isHistoryRecord(v: unknown): v is HistoryRecord {
  if (typeof v !== 'object' || v === null) return false;
  const o = v as Record<string, unknown>;
  return (
    (o.difficulty === 'easy' || o.difficulty === 'normal' || o.difficulty === 'hard' || o.difficulty === 'hell') &&
    typeof o.seed === 'number' &&
    typeof o.timeMs === 'number' &&
    typeof o.steps === 'number' &&
    typeof o.at === 'number' &&
    typeof o.trail === 'string'
  );
}

export async function loadMazeHistory(): Promise<HistoryRecord[]> {
  try {
    const raw = await AsyncStorage.getItem(STORAGE_HISTORY);
    if (!raw) return [];
    const parsed: unknown = JSON.parse(raw);
    if (!Array.isArray(parsed)) return [];
    return parsed.filter(isHistoryRecord);
  } catch {
    return [];
  }
}

/** 新增一条（置顶），返回更新后的列表 */
export async function addHistoryRecord(rec: HistoryRecord): Promise<HistoryRecord[]> {
  const list = await loadMazeHistory();
  const next = [rec, ...list].slice(0, HISTORY_LIMIT);
  await AsyncStorage.setItem(STORAGE_HISTORY, JSON.stringify(next));
  return next;
}

export async function removeHistoryRecord(at: number): Promise<HistoryRecord[]> {
  const next = (await loadMazeHistory()).filter((r) => r.at !== at);
  await AsyncStorage.setItem(STORAGE_HISTORY, JSON.stringify(next));
  return next;
}

export async function clearMazeHistory(): Promise<void> {
  await AsyncStorage.removeItem(STORAGE_HISTORY);
}
```

```tsx
import Svg, { Circle, G, Path, Rect } from 'react-native-svg';
import { FLAG_COLOR, FLAG_POLE_COLOR, START_COLOR, WALL_THICKNESS, type Palette } from '../constants';
import type { MazeData } from '../types';

interface Props {
  maze: MazeData;
  palette: Palette;
  /** 每单元绘制像素（dp）。调用方传入（通常为 BASE_CELL_DP），缩放由容器 transform/澄清机制处理 */
  cellPx: number;
}

/** 静态层：墙糖块 + 起点虚线环 + 终点旗帜。memo 锁死，仅换局/澄清缩放时重绘 */
function MazeSvgInner({ maze, palette, cellPx }: Props) {
  const viewSize = maze.G * cellPx;

  const wallRects = useMemo(() => {
    const grid = maze.G;
    const t = cellPx * WALL_THICKNESS;
    const rects: React.ReactElement[] = [];
    // 8 邻（含对角）：转角/桥接墙格纳入后，双向 run 合并才能拼出连续墙线；
    // 同时保证与碰撞判定一致（cells=1 的墙格都可见，无"看不见的墙"）
    const isWall = (r: number, c: number): boolean => {
      if (maze.cells[r][c] !== 1) return false;
      if (maze.inS[r][c]) return true;
      for (let dr = -1; dr <= 1; dr++) {
        for (let dc = -1; dc <= 1; dc++) {
          if (dr === 0 && dc === 0) continue;
          const nr = r + dr;
          const nc = c + dc;
          if (nr >= 0 && nr < grid && nc >= 0 && nc < grid && maze.inS[nr][nc]) return true;
        }
      }
      return false;
    };
    // 水平墙线：偶数行（水平隔断带，含角位）按 run 合并为细长条（y 取行中心）
    for (let r = 0; r < grid; r += 2) {
      let c = 0;
      while (c < grid) {
        if (!isWall(r, c)) {
          c++;
          continue;
        }
        let end = c;
        while (end < grid && isWall(r, end)) end++;
        rects.push(
          <Rect
            key={`h${r}-${c}`}
            x={c * cellPx}
            y={(r + 0.5) * cellPx - t / 2}
            width={(end - c) * cellPx}
            height={t}
            fill={palette.wall}
          />,
        );
        c = end;
      }
    }
    // 竖直墙线：偶数列（竖直隔断带，含角位）按 run 合并为细长条（x 取列中心）
    for (let c = 0; c < grid; c += 2) {
      let r = 0;
      while (r < grid) {
        if (!isWall(r, c)) {
          r++;
          continue;
        }
        let end = r;
        while (end < grid && isWall(end, c)) end++;
        rects.push(
          <Rect
            key={`v${c}-${r}`}
            x={(c + 0.5) * cellPx - t / 2}
            y={r * cellPx}
            width={t}
            height={(end - r) * cellPx}
            fill={palette.wall}
          />,
        );
        r = end;
      }
    }
    return rects;
  }, [maze, palette.wall, cellPx]);

  const startX = (2 * maze.start[0] + 1.5) * cellPx;
  const startY = (2 * maze.start[1] + 1.5) * cellPx;
  const endX = (2 * maze.end[0] + 1.5) * cellPx;
  const endY = (2 * maze.end[1] + 1.5) * cellPx;
  const flagTop = endY - cellPx * 0.88;
  const flagBot = endY + cellPx * 0.84;
  const flagMid = endY - cellPx * 0.54;
  const flagLow = endY - cellPx * 0.2;

  return (
    <Svg width={viewSize} height={viewSize} viewBox={`0 0 ${viewSize} ${viewSize}`}>
      <Rect x={0} y={0} width={viewSize} height={viewSize} fill={palette.bg} />
      <G>{wallRects}</G>
      <Circle
        cx={startX}
        cy={startY}
        r={cellPx * 0.72}
        fill="none"
        stroke={START_COLOR}
        strokeWidth={cellPx * 0.2}
        strokeDasharray={`${cellPx * 0.39} ${cellPx * 0.32}`}
        strokeLinecap="round"
      />
      <Circle cx={endX} cy={endY} r={cellPx * 1.1} fill="#ffffff" opacity={0.55} />
      <Path
        d={`M ${endX} ${flagBot} L ${endX} ${flagTop}`}
        stroke={FLAG_POLE_COLOR}
        strokeWidth={cellPx * 0.14}
        strokeLinecap="round"
      />
      <Path
        d={`M ${endX} ${flagTop} L ${endX + cellPx * 0.88} ${flagMid} L ${endX} ${flagLow} Z`}
        fill={FLAG_COLOR}
        stroke={FLAG_COLOR}
        strokeWidth={cellPx * 0.12}
        strokeLinejoin="round"
      />
    </Svg>
  );
}

export default React.memo(MazeSvgInner);
```

```tsx
import React, { useEffect, useMemo, useRef, useState } from 'react';
import { StyleSheet, View } from 'react-native';
import type { LayoutChangeEvent } from 'react-native';
import { BASE_CELL_DP, INITIAL_CELL_DP } from '../constants';
import MazeSvg from '../components/MazeSvg';
import HudBar from '../components/HudBar';
import { generateMaze } from '../services/generator';
import { loadBest } from '../services/storage';
import { useMazeStore } from '../store';
import { paletteForSeed } from '../constants';
import type { MazeDifficulty } from '../types';

function randSeed(): number {
  return 1000000 + Math.floor(Math.random() * 9000000);
}

function fmtTime(ms: number): string {
  const total = Math.round(ms / 100) / 10; // 先量化到 0.1s，避免 59.95s 舍入出 "60.0"
  const m = Math.floor(total / 60);
  const s = total - m * 60;
  return `${String(m).padStart(2, '0')}:${s.toFixed(1).padStart(4, '0')}`;
}

export default function MazeGameScreen() {
  const difficulty = useMazeStore((s) => s.difficulty);
  const seed = useMazeStore((s) => s.seed);
  const steps = useMazeStore((s) => s.steps);
  const phase = useMazeStore((s) => s.phase);
  const elapsedMs = useMazeStore((s) => s.elapsedMs);
  const best = useMazeStore((s) => s.best);
  const setDifficulty = useMazeStore((s) => s.setDifficulty);
  const setSeed = useMazeStore((s) => s.setSeed);
  const resetGame = useMazeStore((s) => s.resetGame);
  const setBest = useMazeStore((s) => s.setBest);

  const [viewport, setViewport] = useState({ width: 0, height: 0 });
  const [resetToken, setResetToken] = useState(0);
  const [nowTick, setNowTick] = useState(0);
  const startedAt = useMazeStore((s) => s.startedAt);

  const maze = useMemo(() => generateMaze(seed, difficulty), [seed, difficulty]);
  const palette = useMemo(() => paletteForSeed(seed), [seed]);

  useEffect(() => {
    loadBest().then(setBest);
  }, [setBest]);

  useEffect(() => {
    if (phase !== 'playing') return;
    const timer = setInterval(() => setNowTick((v) => v + 1), 100);
    return () => clearInterval(timer);
  }, [phase]);

  const timeText =
    phase === 'won'
      ? fmtTime(elapsedMs)
      : startedAt > 0
        ? fmtTime(Date.now() - startedAt)
        : '00:00.0';
  const bestText = best[difficulty] ? fmtTime(best[difficulty]!.timeMs) : '--';

  const onLayout = (e: LayoutChangeEvent) => {
    const { width, height } = e.nativeEvent.layout;
    setViewport({ width, height });
  };

  // 初始镜头：小迷宫全览、大迷宫按基础尺寸跟随（细节在镜头任务完善）
  const fit = viewport.width > 0 ? Math.min(viewport.width, viewport.height) / (maze.G + 1.6) : 12;
  const scale = Math.max(fit, INITIAL_CELL_DP) / BASE_CELL_DP;

  // RN transform 围绕元素中心缩放：居中平移与 scale 无关（T12 引擎接管后按同一定义直写）
  const size = maze.G * BASE_CELL_DP;
  const offsetX = viewport.width / 2 - size / 2;
  const offsetY = viewport.height / 2 - size / 2;

  const pickDifficulty = (d: MazeDifficulty) => {
    if (d === difficulty) return;
    setDifficulty(d);
    setSeed(randSeed());
    resetGame();
    setResetToken((v) => v + 1);
  };

  // resetToken：T12 引擎的"重启本局"信号（当前占位，接入后由引擎消费）
  // nowTick：计时显示的渲染驱动（每 100ms 触发 timeText 重新求值）
  void resetToken;
  void nowTick;

  return (
    <View style={[styles.root, { backgroundColor: palette.bg }]}>
      <HudBar
        difficulty={difficulty}
        seed={seed}
        steps={steps}
        timeText={timeText}
        bestText={bestText}
        palette={palette}
        onPickDifficulty={pickDifficulty}
        onPressSeed={() => { /* T16：打开种子面板 */ }}
      />
      <View style={styles.viewport} onLayout={onLayout}>
        {viewport.width > 0 && (
          <View
            style={[
              styles.world,
              { transform: [{ translateX: offsetX }, { translateY: offsetY }, { scale }] },
            ]}>
            <MazeSvg maze={maze} palette={palette} cellPx={BASE_CELL_DP} />
          </View>
        )}
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  root: { flex: 1 },
  viewport: { flex: 1, overflow: 'hidden' },
  world: { position: 'absolute', left: 0, top: 0 },
});
```

```tsx
import React, { useCallback, useMemo, useRef } from 'react';
import { PanResponder, StyleSheet, View } from 'react-native';
import type { GestureResponderEvent } from 'react-native';
import type { JoyVector } from '../types';

interface Props {
  onMove: (v: JoyVector) => void;
  /** 底座直径（建议保持默认 128；旋钮与半径按此尺寸设计） */
  size?: number;
}

/** 自研摇杆：拖动旋钮输出归一化向量 [-1,1]（直写实现，不触发 React 渲染） */
export default function Joystick({ onMove, size = 128 }: Props) {
  const knobRef = useRef<View>(null);
  const baseRef = useRef<View>(null);
  const originRef = useRef({ x: 0, y: 0 });
  const onMoveRef = useRef(onMove);
  onMoveRef.current = onMove;
  const radius = size / 2 - 26;

  const measure = useCallback(() => {
    baseRef.current?.measureInWindow((x, y) => {
      originRef.current = { x, y };
    });
  }, []);

  const responder = useMemo(
    () =>
      PanResponder.create({
        onStartShouldSetPanResponder: () => true,
        onMoveShouldSetPanResponder: () => true,
        onPanResponderGrant: (evt) => handle(evt),
        onPanResponderMove: (evt) => handle(evt),
        onPanResponderRelease: reset,
        onPanResponderTerminate: reset,
      }),
    // eslint-disable-next-line react-hooks/exhaustive-deps
    [],
  );

  function handle(evt: GestureResponderEvent) {
    const px = evt.nativeEvent.pageX;
    const py = evt.nativeEvent.pageY;
    let dx = px - originRef.current.x - size / 2;
    let dy = py - originRef.current.y - size / 2;
    const mag = Math.hypot(dx, dy);
    if (mag > radius) {
      dx = (dx / mag) * radius;
      dy = (dy / mag) * radius;
    }
    knobRef.current?.setNativeProps({

    });
    onMoveRef.current({ x: dx / radius, y: dy / radius });
  }

  function reset() {
    knobRef.current?.setNativeProps({ style: { transform: [{ translateX: 0 }, { translateY: 0 }] } });
    onMoveRef.current({ x: 0, y: 0 });
  }

  return (
    <View
      testID="joy-base"
      ref={baseRef}
      onLayout={measure}
      style={[styles.base, { width: size, height: size, borderRadius: size / 2 }]}
      {...responder.panHandlers}>
      <View
        testID="joy-knob"
        ref={knobRef}
        style={[styles.knob, { borderRadius: (size - 70) / 2 + 29 }]}
      />
    </View>
  );
```

```ts
import { useEffect, useRef } from 'react';
import type { View } from 'react-native';
import { BALL_R, END_TOL, JOY_DEAD, SPEED, BASE_CELL_DP } from '../constants';
import { moveBall } from '../services/collision';
import type { BallPos, CellPoint, JoyVector, MazeData } from '../types';

export interface EngineCallbacks {
  /** 小球首次移动（开始计时） */
  onStart: () => void;
  /** 进入新格（含起点外每步） */
  onEnterCell: (cell: CellPoint) => void;
  /** 通关（elapsedMs 为首次移动至今） */
  onWin: (elapsedMs: number) => void;
}

export interface EngineOptions extends EngineCallbacks {
  maze: MazeData;
  joyRef: React.MutableRefObject<JoyVector>;
  ballViewRef: React.RefObject<View | null>;
  containerRef: React.RefObject<View | null>;
  viewport: { width: number; height: number };
  /** 变化时重置一局（重开/换局） */
  resetToken: number;
  /** 镜头：相对基准格的缩放倍数（Task 15 会由此 hook 接管计算） */
  camScaleRef: React.MutableRefObject<number>;
}

/**
 * 引擎职责：
 * - requestAnimationFrame 循环读取摇杆向量 → moveBall 物理 → 步数/通关事件
 * - 小球视图直写（位置含镜头变换）
 * - 容器平移直写（镜头跟随，Task 15 完善手动模式）
 */
export function useGameEngine(opts: EngineOptions) {
  const { maze, joyRef, ballViewRef, containerRef, viewport, resetToken, camScaleRef } = opts;
  const cbRef = useRef(opts);
  cbRef.current = opts;

  const ball = useRef<BallPos>({ x: 0, y: 0 });
  const curCell = useRef<CellPoint>([0, 0]);
  const phaseRef = useRef<'ready' | 'playing' | 'won'>('ready');
  const startTimeRef = useRef(0);
  const cam = useRef({ x: maze.G / 2, y: maze.G / 2 });
  const rafRef = useRef(0);
  const lastTsRef = useRef(0);
  // 终点世界坐标（每渲染重算；tick 闭包随 effect 依赖 maze 重建）
  const endPos = { x: 2 * maze.end[0] + 1.5, y: 2 * maze.end[1] + 1.5 };

  function camScale(): number {
    return camScaleRef.current;
  }

  /** 世界坐标 → 屏幕坐标 */
  function worldToScreen(x: number, y: number): { sx: number; sy: number } {
    const s = camScale();
    const k = BASE_CELL_DP * s;
    return {
      sx: (x - cam.current.x) * k + viewport.width / 2,
      sy: (y - cam.current.y) * k + viewport.height / 2,
    };
  }

  function writeBall() {
    const v = ballViewRef.current;
    if (!v) return;
    const { sx, sy } = worldToScreen(ball.current.x, ball.current.y);
    const d = BALL_R * 2 * BASE_CELL_DP * camScale();
    v.setNativeProps({
      style: {
        width: d,
        height: d,
        borderRadius: d / 2,
        transform: [{ translateX: sx - d / 2 }, { translateY: sy - d / 2 }],
      },
    });
  }

  function writeContainer() {
    const v = containerRef.current;
    if (!v) return;
    const s = camScale();
    // RN transform 围绕元素中心缩放：先居中平移（与 s 无关），
    // 再按 s 反向补偿镜头中心相对元素中心的偏移
    const size = maze.G * BASE_CELL_DP;
    const tx =
      viewport.width / 2 - size / 2 - s * (cam.current.x * BASE_CELL_DP - size / 2);
    const ty =
      viewport.height / 2 - size / 2 - s * (cam.current.y * BASE_CELL_DP - size / 2);
    v.setNativeProps({
      style: { transform: [{ translateX: tx }, { translateY: ty }, { scale: s }] },
    });
  }

  function reset() {
    // 原地更新：保证 getBall/getCam 已流出的引用不失效（调用方可能缓存）
    ball.current.x = 2 * maze.start[0] + 1.5;
    ball.current.y = 2 * maze.start[1] + 1.5;
    curCell.current = [maze.start[0], maze.start[1]];
    phaseRef.current = 'ready';
    startTimeRef.current = 0;
    cam.current.x = maze.G / 2;
    cam.current.y = maze.G / 2;
    writeBall();
    writeContainer();
  }

  useEffect(() => {
    reset();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [maze, resetToken]);

  // 视口尺寸变化（旋转/分屏）：仅重新布局，不重置对局
  useEffect(() => {
    if (viewport.width <= 0) return;
    writeBall();
    writeContainer();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [viewport.width, viewport.height]);

  useEffect(() => {
    function tick(ts: number) {
      const dt = lastTsRef.current ? Math.min((ts - lastTsRef.current) / 1000, 0.033) : 0;
      lastTsRef.current = ts;
      if (phaseRef.current !== 'won' && viewport.width > 0) {
        const j = joyRef.current;
        const mag = Math.hypot(j.x, j.y);
        if (mag > JOY_DEAD) {
          if (phaseRef.current === 'ready') {
            phaseRef.current = 'playing';
            startTimeRef.current = Date.now();
            cbRef.current.onStart();
          }
          const amp = Math.pow((mag - JOY_DEAD) / (1 - JOY_DEAD), 1.15);
          const ux = j.x / mag;
          const uy = j.y / mag;
          moveBall(maze, ball.current, ux * SPEED * amp * dt, uy * SPEED * amp * dt);
          const ci = Math.floor((ball.current.x - 1) / 2);
          const cj = Math.floor((ball.current.y - 1) / 2);
          if (ci !== curCell.current[0] || cj !== curCell.current[1]) {
            curCell.current = [ci, cj];
            cbRef.current.onEnterCell([ci, cj]);
          }
          if (Math.hypot(ball.current.x - endPos.x, ball.current.y - endPos.y) < END_TOL) {
            phaseRef.current = 'won';
            cbRef.current.onWin(Date.now() - startTimeRef.current);
          }
        }
        updateCamera(dt);
        writeBall();
        writeContainer();
      }
      rafRef.current = requestAnimationFrame(tick);
    }
    rafRef.current = requestAnimationFrame(tick);
    return () => cancelAnimationFrame(rafRef.current);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [maze, viewport.width, viewport.height]);

  /** 跟随镜头：死区 30%，出界平滑追踪（手动模式的接管在 Task 15） */
  function updateCamera(dt: number) {
    const s = camScale();
    const wW = viewport.width / (BASE_CELL_DP * s);
    const wH = viewport.height / (BASE_CELL_DP * s);
    if (maze.G <= wW && maze.G <= wH) {
      cam.current.x = maze.G / 2;
      cam.current.y = maze.G / 2;
      return;
    }
    const dx = ball.current.x - cam.current.x;
    const dy = ball.current.y - cam.current.y;
    const deadX = wW * 0.3;
    const deadY = wH * 0.3;
    let tx = cam.current.x;
    let ty = cam.current.y;
    if (dx > deadX) tx = ball.current.x - deadX;
    if (dx < -deadX) tx = ball.current.x + deadX;
    if (dy > deadY) ty = ball.current.y - deadY;
    if (dy < -deadY) ty = ball.current.y + deadY;
    const k = Math.min(1, dt * 7);
    cam.current.x += (tx - cam.current.x) * k;
    cam.current.y += (ty - cam.current.y) * k;
    // 镜头不越出迷宫
    cam.current.x = Math.max(wW / 2, Math.min(maze.G - wW / 2, cam.current.x));
    cam.current.y = Math.max(wH / 2, Math.min(maze.G - wH / 2, cam.current.y));
  }

  return {
    /** 供外部读取当前球位置（幽灵/缩略图） */
    getBall: () => ball.current,
    getCam: () => cam.current,
  };
}
```

```tsx
import React, { useMemo } from 'react';
import Svg, { Path } from 'react-native-svg';
import { FADE_CUTOFF, FADE_DECAY, GHOST_COLOR, PATH_COLOR, TRAIL_WIDTH, type Palette } from '../constants';
import type { CellPoint, MazeData, TrailStyle } from '../types';

interface Props {
  maze: MazeData;
  cellPx: number;
  palette: Palette;
  trailStyle: TrailStyle;
  trail: CellPoint[];
  ghost: CellPoint[] | null;
  shortestPath: CellPoint[] | null;

function hslToHex(h: number, s: number, l: number): string {
  const c = (1 - Math.abs(2 * l - 1)) * s;
  const hp = h / 60;
  const x = c * (1 - Math.abs((hp % 2) - 1));
  let rgb: [number, number, number];
  if (hp < 1) rgb = [c, x, 0];
  else if (hp < 2) rgb = [x, c, 0];
  else if (hp < 3) rgb = [0, c, x];
  else if (hp < 4) rgb = [0, x, c];
  else if (hp < 5) rgb = [x, 0, c];
  else rgb = [c, 0, x];
  const m = l - c / 2;
  const toHex = (v: number) =>
    Math.round((v + m) * 255)
      .toString(16)
      .padStart(2, '0');
  return `#${toHex(rgb[0])}${toHex(rgb[1])}${toHex(rgb[2])}`;
}

/** 12 个彩虹桶中心色（hue = 15 + 30k） */
const RAINBOW_BUCKETS: string[] = Array.from({ length: 12 }, (_, k) => hslToHex(15 + 30 * k, 0.85, 0.52));
const FADE_STEPS = 8;

function cellPoint(cell: CellPoint, cellPx: number): string {
  return `${(2 * cell[0] + 1.5) * cellPx} ${(2 * cell[1] + 1.5) * cellPx}`;
}

function buildSegments(trail: CellPoint[], cellPx: number): string[] {
  const segs: string[] = [];
  for (let n = 1; n < trail.length; n++) {
    segs.push(`M ${cellPoint(trail[n - 1], cellPx)} L ${cellPoint(trail[n], cellPx)}`);
  }
  return segs;
}

function TrailLayerInner({ maze, cellPx, palette, trailStyle, trail, ghost, shortestPath }: Props) {
  const size = maze.G * cellPx;

  const paths = useMemo(() => {
    if (trail.length < 2) return [] as { d: string; color: string; opacity: number }[];
    const segs = buildSegments(trail, cellPx);
    if (trailStyle === 'mono') {
      return [{ d: segs.join(' '), color: palette.wallDark, opacity: 1 }];
    }
    if (trailStyle === 'fade') {
      // 单色 8 档透明度：越旧越淡，低于阈值直接丢弃
      const buckets: string[][] = Array.from({ length: FADE_STEPS }, () => []);
      const total = segs.length;
      for (let n = 0; n < total; n++) {
        const age = total - 1 - n;
        const alpha = Math.pow(FADE_DECAY, age);
        if (alpha < FADE_CUTOFF) continue;
        const step = Math.min(FADE_STEPS - 1, Math.floor(alpha * FADE_STEPS));
        buckets[step].push(segs[n]);
      }
      return buckets
        .map((list, step) => ({
          d: list.join(' '),
          color: palette.wallDark,
          opacity: (step + 1) / FADE_STEPS,
        }))
        .filter((p) => p.d.length > 0);
    }
    // rainbow：hue = (步序 × 15) % 360 → 12 桶（色桶中心取 30° 台阶）
    const buckets: string[][] = Array.from({ length: 12 }, () => []);
    for (let n = 0; n < segs.length; n++) {
      const hue = ((n + 1) * 15) % 360;
      buckets[Math.floor(hue / 30) % 12].push(segs[n]);
    }
    return buckets
      .map((list, k) => ({ d: list.join(' '), color: RAINBOW_BUCKETS[k], opacity: 1 }))
      .filter((p) => p.d.length > 0);
  }, [trail, cellPx, trailStyle, palette.wallDark]);

  const ghostD = useMemo(() => (ghost ? buildSegments(ghost, cellPx).join(' ') : ''), [ghost, cellPx]);
  const pathD = useMemo(
    () => (shortestPath ? buildSegments(shortestPath, cellPx).join(' ') : ''),
    [shortestPath, cellPx],
  );

  return (
    <Svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
      {ghostD !== '' && (
        <Path d={ghostD} stroke={GHOST_COLOR} strokeWidth={cellPx * 0.2} strokeLinecap="round" strokeLinejoin="round" fill="none" />
      )}
      {pathD !== '' && (
        <Path
          d={pathD}
          stroke={PATH_COLOR}
          strokeWidth={cellPx * 0.16}
          strokeLinecap="round"
          strokeDasharray={`${cellPx * 0.42} ${cellPx * 0.42}`}
          fill="none"
        />
      )}
      {paths.map((p, idx) => (
        <Path
          key={idx}
          d={p.d}
          stroke={p.color}
          strokeWidth={cellPx * TRAIL_WIDTH}
          strokeLinecap="round"
          strokeLinejoin="round"
          strokeOpacity={p.opacity}
          fill="none"
        />
      ))}
    </Svg>
  );
}

export default React.memo(TrailLayerInner);
```

```ts
/** 格坐标 BFS 最短路；无路径返回 null */
export function findShortestPath(maze: MazeData, from: CellPoint, to: CellPoint): CellPoint[] | null {
  const keyOf = (i: number, j: number) => i + ',' + j;
  const seen = new Set<string>([keyOf(from[0], from[1])]);
  const q: CellPoint[] = [from];
  const prev: Record<string, CellPoint> = {};
  const dirs = [
    [1, 0],
    [-1, 0],
    [0, 1],
    [0, -1],
  ];
  let found = false;
  while (q.length) {
    const cur = q.shift()!;
    if (cur[0] === to[0] && cur[1] === to[1]) {
      found = true;
      break;
    }
    for (const [di, dj] of dirs) {
      const ni = cur[0] + di;
      const nj = cur[1] + dj;
      if (ni < 0 || nj < 0 || ni >= maze.N || nj >= maze.N) continue;
      if (maze.cells[2 * nj + 1][2 * ni + 1] !== 0) continue;
      if (maze.cells[2 * cur[1] + 1 + dj][2 * cur[0] + 1 + di] !== 0) continue;
      const k = keyOf(ni, nj);
      if (seen.has(k)) continue;
      seen.add(k);
      prev[k] = cur;
      q.push([ni, nj]);
    }
  }
  if (!found) return null;
  const path: CellPoint[] = [];
  let node: CellPoint | undefined = [to[0], to[1]];
  while (node) {
    path.unshift(node);
    node = prev[keyOf(node[0], node[1])];
  }
  return path;
}
```

```tsx
      onStartShouldSetPanResponder: () => true,
      onMoveShouldSetPanResponder: () => true,
      onPanResponderGrant: (evt) => {
        // 清理上一轮手势尚未触发的澄清定时器（防陈旧 settle 覆盖新镜头）
        if (settleTimerRef.current) {
          clearTimeout(settleTimerRef.current);
          settleTimerRef.current = null;
        }
        const now = Date.now();
        // 双击 = 两次 grant 间隔 <300ms 且期间无移动（防吞掉带小位移的手势）
        if (now - lastTapRef.current < 300 && !movedSinceGrantRef.current) {
          engineRef.current.followBall();
          lastTapRef.current = 0;
          dragRef.current = null;
          pinchRef.current = null;
          movedSinceGrantRef.current = false;
          return;
        }
        lastTapRef.current = now;
        movedSinceGrantRef.current = false;
        const touches = evt.nativeEvent.touches;
        if (touches.length >= 2) {
          pinchRef.current = {
            d: Math.hypot(touches[0].pageX - touches[1].pageX, touches[0].pageY - touches[1].pageY),
          };
        } else if (touches.length === 1) {
          dragRef.current = { x: touches[0].pageX, y: touches[0].pageY };
        }
      },
      onPanResponderMove: (evt) => {
        movedSinceGrantRef.current = true;
        const touches = evt.nativeEvent.touches;
        if (touches.length >= 2) {
          if (!pinchRef.current) {
            // 第二指刚落下：以当前两指距离为基准启动捏合，并中止拖动
            pinchRef.current = {
              d: Math.hypot(
                touches[0].pageX - touches[1].pageX,
                touches[0].pageY - touches[1].pageY,
              ),
            };
            dragRef.current = null;
            movedSinceGrantRef.current = true;
          }
          const d = Math.hypot(
            touches[0].pageX - touches[1].pageX,
            touches[0].pageY - touches[1].pageY,
          );
          if (pinchRef.current.d > 0) {
            engineRef.current.zoomAt(
              (touches[0].pageX + touches[1].pageX) / 2 - viewportOriginRef.current.x,
              (touches[0].pageY + touches[1].pageY) / 2 - viewportOriginRef.current.y,
              d / pinchRef.current.d,
            );
          }
          pinchRef.current.d = d;
          engineRef.current.manual();
        } else if (touches.length === 1) {
          if (pinchRef.current) {
            // 松开一指回到单指：重置捏合，把剩余手指作为拖动起点
            pinchRef.current = null;
            dragRef.current = { x: touches[0].pageX, y: touches[0].pageY };
            return;
          }
          if (dragRef.current) {
            const dx = touches[0].pageX - dragRef.current.x;
            const dy = touches[0].pageY - dragRef.current.y;
            engineRef.current.pan(dx, dy);
            dragRef.current = { x: touches[0].pageX, y: touches[0].pageY };
            engineRef.current.manual();
          }
        }
      },
      onPanResponderRelease: endGesture,
      onPanResponderTerminate: endGesture,
    }),
    [],
  );

function endGesture() {
  dragRef.current = null;
  pinchRef.current = null;
  if (settleTimerRef.current) clearTimeout(settleTimerRef.current);
  settleTimerRef.current = setTimeout(() => {
    settleTimerRef.current = null;
    const px = engineRef.current.settleZoom();
    if (px) setCellPxState(px);
  }, 250);
}
```

- [ ] **Step 5: 引擎生命周期加固（后台/焦点感知）**

为 `useGameEngine` 增加 AppState 感知：切后台停止 rAF 调度（避免空转与向 detach 视图直写）、回前台恢复。实际实现要点：

1. `tick` 从 `useEffect` 内**提升为 hook 体的普通函数**，并增加 `tickRef`——rAF 链经 `tickRef.current` 自续，任何时刻都使用最新渲染闭包（maze/viewport 变化无需等 effect 重建生效）；
2. tick 首行检查 `appActiveRef.current`：非活跃时置 `rafRef.current = 0` 并直接返回（不再调度下一帧）；
3. tick 的 effect（依赖 `[maze, viewport.width, viewport.height]`）只负责启动与清理：启动前 `lastTsRef.current = 0`，清理时 `cancelAnimationFrame(rafRef.current)` + `rafRef.current = 0`（其内已无闭包变量引用，不再需要 exhaustive-deps disable 注释）；
4. 新增独立 AppState effect（依赖 `[]`）：监听 `change` 更新 `appActiveRef`；回前台（`!wasActive`）**无条件重置 `lastTsRef = 0`**（避免大 dt 瞬移；即使 rAF 链仍排队——被冻结的首帧未到——恢复首帧的 dt 也为 0，避免 33ms 钳制造成单帧跳步），仅当 `rafRef.current === 0` 时重新调度 rAF（防重复排队叠加链）；卸载时 `sub.remove()`。

```ts
const appActiveRef = useRef(true);
const tickRef = useRef<(ts: number) => void>(() => {});
function tick(ts: number) {
  if (!appActiveRef.current) {
    rafRef.current = 0;
    return;
  }
  // ...原 tick 全部逻辑...
  rafRef.current = requestAnimationFrame(tickRef.current);
}
tickRef.current = tick;
// 双 effect：tick effect（[maze, viewport.width, viewport.height]）启动/清理；
// AppState effect（[]）change → appActiveRef 更新；回前台（!wasActive）无条件
//   lastTsRef.current = 0，仅 rafRef.current === 0 时 requestAnimationFrame(tickRef.current)
```

测试：`useGameEngine.test.ts` 追加 AppState 生命周期用例（`jest.spyOn(AppState, 'addEventListener')` 捕获监听手动触发）：

1. 「AppState 切后台后停止 rAF，回前台恢复并重置时间基准」（含 10s 时间跳跃帧验证 dt=0 不瞬移）；
2. 「重复 active 事件不叠加 rAF 链」（连续 active 后 rAF 队列长度恒为 1）；
3. 「inactive/background → active 循环后恢复单链」（background 下不消费帧、`rafRef!==0` 时回前台：10s 跳跃恢复首帧 dt=0 不跳步，随后 16ms 帧正常步进）；
4. 「卸载时移除 AppState 监听」（unmount 后 `sub.remove()` 恰好调用一次）。

变异验证：去掉后台早退、去掉 lastTs 重置均被用例捕获；把无条件重置改回「仅 `rafRef===0` 时重置」后，用例 3 的球位置断言失败（恢复首帧 dt 被钳到 33ms，球单帧跳步约 0.36 世界单位，可捕获）。

真机验证：切后台 10 秒回前台，游戏状态与计时连续（计时因 startedAt 基准为 Date.now 会包含后台时间——如不符预期改为累计法，届时决策）。
